function retain(source: string, label: string, expected?: string) {
 const bytes = readFileSync(source); if (expected) assert.equal(hash(bytes), expected, source)
 const zipped = bytes.length > 262144, name = label + (zipped ? '.gz' : ''), output = zipped ? gzipSync(bytes) : bytes

 files.push(item); return name
}
